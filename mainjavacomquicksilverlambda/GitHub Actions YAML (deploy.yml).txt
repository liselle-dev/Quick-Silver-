GitHub Actions YAML (deploy.yml)


name: Quicksilver Java Lambda Deployment

on:
  push:
    branches:
      - main

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      # Step 1: Pull your code
      - name: Checkout code
        uses: actions/checkout@v4

      # Step 2: Setup Java 17
      - name: Setup Java
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'

      # Step 3: Build project with Maven
      - name: Build with Maven
        run: mvn clean package

      # Step 4: Configure AWS credentials
      - name: Configure AWS Credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: us-east-2

      # Step 5: Deploy your Lambda functions automatically
      - name: Deploy Payment Lambda
        run: |
          aws lambda update-function-code \
            --function-name QuickSilverPaymentLambda \
            --zip-file fileb://target/quicksilver.jar

      - name: Deploy Signup Lambda
        run: |
          aws lambda update-function-code \
            --function-name QuickSilverSignupLambda \
            --zip-file fileb://target/quicksilver.jar

      - name: Deploy Edit Profile Lambda
        run: |
          aws lambda update-function-code \
            --function-name QuickSilverEditProfileLambda \
            --zip-file fileb://target/quicksilver.jar

      - name: Deploy Main Menu Lambda
        run: |
          aws lambda update-function-code \
            --function-name QuickSilverMainMenuLambda \
            --zip-file fileb://target/quicksilver.jar
