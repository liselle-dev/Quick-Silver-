LambdaSignUp_Function


import json
import os
import boto3
from botocore.exceptions import ClientError
from datetime import datetime

# Environment variables
REGION = os.getenv("REGION")
USER_POOL_ID = os.getenv("COGNITO_USER_POOL_ID")
APP_CLIENT_ID = os.getenv("COGNITO_APP_CLIENT_ID")
USERS_TABLE = os.getenv("USERS_TABLE")

# AWS clients
cognito = boto3.client("cognito-idp", region_name=REGION)
dynamodb = boto3.resource("dynamodb", region_name=REGION)
users_table = dynamodb.Table(USERS_TABLE)


def lambda_handler(event, context):
    try:
        # Parse body (API Gateway)
        if "body" in event:
            body = json.loads(event["body"])
        else:
            body = event

        email = body.get("email")
        password = body.get("password")
        name = body.get("name")

        if not email or not password or not name:
            return {
                "statusCode": 400,
                "body": json.dumps({"message": "Email, password, and name are required"})
            }

        # 1️⃣ Create user in Cognito
        cognito.sign_up(
            ClientId=APP_CLIENT_ID,
            Username=email,
            Password=password,
            UserAttributes=[
                {"Name": "email", "Value": email},
                {"Name": "name", "Value": name}
            ]
        )

        # 2️⃣ Auto-confirm user (optional but common in server-side signup)
        cognito.admin_confirm_sign_up(
            UserPoolId=USER_POOL_ID,
            Username=email
        )

        # 3️⃣ Create user record in DynamoDB
        users_table.put_item(
            Item={
                "userId": email,
                "name": name,
                "createdAt": datetime.utcnow().isoformat() + "Z"
            }
        )

        return {
            "statusCode": 201,
            "body": json.dumps({"message": "User registration successful"})
        }

    except ClientError as e:
        print("Cognito Error:", e)
        return {
            "statusCode": 500,
            "body": json.dumps({"message": "Signup failed", "error": str(e)})
        }

    except Exception as e:
        print("General Error:", e)
        return {
            "statusCode": 500,
            "body": json.dumps({"message": "Signup failed", "error": str(e)})
        }
