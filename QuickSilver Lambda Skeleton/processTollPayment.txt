import json
import boto3
import os
from datetime import datetime
from decimal import Decimal

AWS_REGION = os.environ.get("AWS_REGION", "us-east-1")

dynamodb = boto3.resource("dynamodb", region_name=AWS_REGION)
sns_client = boto3.client("sns", region_name=AWS_REGION)

# Replaced environment variable with actual ARN
SNS_TOPIC_ARN = "arn:aws:sns:us-east-2:142754574142:TollPaymentNotifications"

TRIPS_TABLE_NAME = os.environ.get("TRIPS_TABLE", "QuickSilver-Trips")
PAYMENTS_TABLE_NAME = os.environ.get("PAYMENTS_TABLE", "QuickSilver-PaymentMethods")
NOTIFICATIONS_TABLE_NAME = os.environ.get("NOTIFICATIONS_TABLE", "QuickSilver-Notifications")


def lambda_handler(event, context):
    print("Received payment event:", json.dumps(event))

    try:
        # ---------------------------
        # Parse request body
        # ---------------------------
        body = json.loads(event["body"]) if "body" in event and isinstance(event["body"], str) else event

        trip_id = body["tripId"]
        user_id = body["userId"]
        payment_method_id = body["paymentMethodId"]

        trips_table = dynamodb.Table(TRIPS_TABLE_NAME)
        payments_table = dynamodb.Table(PAYMENTS_TABLE_NAME)
        notifications_table = dynamodb.Table(NOTIFICATIONS_TABLE_NAME)

        # ---------------------------
        # 1️⃣ Get Trip
        # ---------------------------
        trip_response = trips_table.get_item(
            Key={"userID": user_id, "id": trip_id}
        )
        trip = trip_response.get("Item")

        if not trip:
            return response(404, {"error": "Trip not found"})

        if trip.get("status") == "paid":
            return response(409, {"error": "Trip already paid"})

        # ---------------------------
        # 2️⃣ Get Payment Method
        # ---------------------------
        pm_response = payments_table.get_item(
            Key={"userID": user_id, "id": payment_method_id}
        )
        payment_method = pm_response.get("Item")

        if not payment_method:
            return response(404, {"error": "Payment method not found"})

        # ---------------------------
        # 3️⃣ Generate Confirmation
        # ---------------------------
        confirmation_number = f"QS{int(datetime.utcnow().timestamp() * 1000)}"
        now_iso = datetime.utcnow().isoformat() + "Z"

        # ---------------------------
        # 4️⃣ Update Trip
        # ---------------------------
        trips_table.update_item(
            Key={"userID": user_id, "id": trip_id},
            UpdateExpression="""
                SET #status = :status,
                    payment_method = :pm,
                    confirmation_number = :conf,
                    updated_date = :updated
            """,
            ExpressionAttributeNames={"#status": "status"},
            ExpressionAttributeValues={
                ":status": "paid",
                ":pm": payment_method["last_four"],
                ":conf": confirmation_number,
                ":updated": now_iso,
                ":paid": "paid"
            },
            ConditionExpression="#status <> :paid"
        )

        # ---------------------------
        # 5️⃣ Create Notification
        # ---------------------------
        notification_id = f"notif-{int(datetime.utcnow().timestamp() * 1000)}"

        notifications_table.put_item(
            Item={
                "id": notification_id,
                "userID": user_id,
                "type": "payment_success",
                "title": "Payment Successful",
                "message": f"Your ${float(trip['amount']):.2f} toll payment was successful.",
                "is_read": False,
                "priority": "medium",
                "trip_id": trip_id,
                "metadata": {
                    "confirmation_number": confirmation_number
                },
                "created_date": now_iso,
                "updated_date": now_iso
            }
        )

        # ---------------------------
        # 6️⃣ SNS Notification (NON-CRITICAL)
        # ---------------------------
        if SNS_TOPIC_ARN:
            try:
                sns_message = (
                    f"Payment Successful!\n"
                    f"Amount: ${float(trip['amount']):.2f}\n"
                    f"Confirmation: {confirmation_number}\n"
                    f"Toll: {trip.get('toll_location')}\n"
                    f"Vehicle: {trip.get('license_plate')}"
                )

                sns_client.publish(
                    TopicArn=SNS_TOPIC_ARN,
                    Message=sns_message,
                    Subject="QuickSilver Payment Confirmation"
                )

            except Exception as sns_error:
                print(f"SNS Error (non-critical): {sns_error}")

        # ---------------------------
        # Success Response
        # ---------------------------
        return response(200, {
            "message": "Payment processed successfully",
            "confirmationNumber": confirmation_number,
            "tripId": trip_id
        })

    # ---------------------------
    # Main Error Handler
    # ---------------------------
    except Exception as error:
        print("Error processing payment:", str(error))
        return response(500, {"error": "Internal server error"})


def response(status_code, body):
    return {
        "statusCode": status_code,
        "body": json.dumps(body)
    }
