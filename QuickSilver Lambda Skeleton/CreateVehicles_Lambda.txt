CreateVehicles_Lambda

import json
import os
import boto3
from datetime import datetime
from botocore.exceptions import ClientError
import logging

# --------------------------- 
# Logging
# ---------------------------
logger = logging.getLogger()
logger.setLevel(logging.INFO)

# --------------------------- 
# AWS clients
# ---------------------------
dynamodb = boto3.resource("dynamodb")
sns = boto3.client("sns")

# --------------------------- 
# Configuration
# ---------------------------
VEHICLES_TABLE = os.environ.get("VEHICLES_TABLE", "QuickSilver-Vehicles")
IDEMPOTENCY_TABLE = os.environ.get("IDEMPOTENCY_TABLE", "QuickSilver-Idempotency")
SNS_TOPIC_ARN = os.environ.get("SNS_TOPIC_ARN")

vehicles_table = dynamodb.Table(VEHICLES_TABLE)
idempotency_table = dynamodb.Table(IDEMPOTENCY_TABLE)

# --------------------------- 
# Helper: Check and record idempotency
# ---------------------------
def is_already_processed(idempotency_key: str, expiration: int = 3600) -> bool:
    """
    Check if this request was already processed.
    Returns True if already processed (and notification sent).
    """
    try:
        response = idempotency_table.get_item(Key={"idempotency_key": idempotency_key})
        return "Item" in response
    except ClientError as e:
        logger.error(f"Error checking idempotency: {e}")
        raise

def record_processed(idempotency_key: str, expiration: int = 3600):
    """
    Record that this request has been processed (SNS sent).
    Set TTL to auto-expire old records.
    """
    ttl = int(datetime.utcnow().timestamp()) + expiration
    try:
        idempotency_table.put_item(
            Item={
                "idempotency_key": idempotency_key,
                "created_at": datetime.utcnow().isoformat(),
                "ttl": ttl
            }
        )
    except ClientError as e:
        logger.error(f"Error recording idempotency: {e}")
        # Don't fail the whole request if idempotency record fails
        pass

# --------------------------- 
# Lambda handler
# ---------------------------
def lambda_handler(event, context):
    try:
        # Parse body
        body = json.loads(event.get("body", "{}")) if isinstance(event.get("body"), str) else event.get("body", {})

        # Required fields
        user_id = body.get("userID")
        vehicle_id = body.get("vehicleID")
        idempotency_key_from_client = body.get("idempotencyKey")  # Optional client-provided key

        if not user_id or not vehicle_id:
            return {
                "statusCode": 400,
                "body": json.dumps({"message": "userID and vehicleID are required"})
            }

        # --------------------------- 
        # Generate idempotency key
        # Use client-provided if available, otherwise fallback to userID + vehicleID
        # ---------------------------
        idempotency_key = idempotency_key_from_client or f"{user_id}#{vehicle_id}"

        # --------------------------- 
        # Check if already processed (prevents duplicate SNS)
        # ---------------------------
        if is_already_processed(idempotency_key):
            logger.info(f"Duplicate request detected for idempotency_key: {idempotency_key}")
            return {
                "statusCode": 200,  # Idempotent success
                "body": json.dumps({"message": "Vehicle already created (idempotent success)"})
            }

        # --------------------------- 
        # Build vehicle item
        # ---------------------------
        vehicle_item = {
            "userID": user_id,
            "vehicleID": vehicle_id,
            "make": body.get("make", ""),
            "model": body.get("model", ""),
            "year": body.get("year", ""),
            "licensePlate": body.get("licensePlate", ""),
            "color": body.get("color", ""),
            "nickname": body.get("nickname", ""),
            "createdAt": datetime.utcnow().isoformat(),
            "updatedAt": datetime.utcnow().isoformat()
        }
